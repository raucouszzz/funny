                applyFirstPersonUnlock()

        toggleUiUpdaters["Unlock First Person"](state)

    if not FullbrightEnabled then return end
    Lighting.Ambient = Color3.new(LightIntensity, LightIntensity, LightIntensity)

        Lighting.FogEnd = 100000
    end

local Ext = {}

do
    local enabled = false
    local token = 0
    local conns = {}
    local savedLighting = nil
    local savedInst = setmetatable({}, { __mode = "k" })

    local function clean(inst)
        pcall(function()
            if inst:IsA("Atmosphere") then
                if not savedInst[inst] then
                    savedInst[inst] = { Density = inst.Density, Haze = inst.Haze }
                end
                if inst.Density ~= 0 then inst.Density = 0 end
                if inst.Haze ~= 0 then inst.Haze = 0 end
            elseif inst:IsA("BlurEffect") or inst:IsA("DepthOfFieldEffect") or inst:IsA("ColorCorrectionEffect") then
                if not savedInst[inst] then
                    savedInst[inst] = { Enabled = inst.Enabled }
                end
                if inst.Enabled then inst.Enabled = false end
            end
        end)
    end

    local function clearFog()
        pcall(function()
            if Lighting.FogEnd < 1000000 then Lighting.FogEnd = 1000000 end
            if Lighting.FogStart ~= 0 then Lighting.FogStart = 0 end
        end)
    end

    local function sweep()
        for _, d in ipairs(Lighting:GetDescendants()) do clean(d) end
        local cam = Workspace.CurrentCamera
        if cam then
            for _, d in ipairs(cam:GetChildren()) do clean(d) end
        end
        clearFog()
    end

    function Ext.toggleRemoveFog(state)
        if state and not enabled then
            enabled = true
            token += 1
            savedLighting = { FogEnd = Lighting.FogEnd, FogStart = Lighting.FogStart }
            sweep()
            table.insert(conns, Lighting.DescendantAdded:Connect(function(d) task.defer(clean, d) end))
            table.insert(conns, Lighting:GetPropertyChangedSignal("FogEnd"):Connect(clearFog))
            table.insert(conns, Lighting:GetPropertyChangedSignal("FogStart"):Connect(clearFog))
            local cam = Workspace.CurrentCamera
            if cam then
                table.insert(conns, cam.ChildAdded:Connect(function(d) task.defer(clean, d) end))
            end
            local myToken = token
            task.spawn(function()
                while myToken == token do
                    task.wait(0.5)
                    if myToken ~= token then break end
                    sweep()
                end
            end)
        elseif not state and enabled then
            enabled = false
            token += 1
            for _, c in ipairs(conns) do c:Disconnect() end
            table.clear(conns)
            for inst, s in pairs(savedInst) do
                if inst.Parent then
                    pcall(function()
                        for k, v in pairs(s) do inst[k] = v end
                    end)
                end
            end
            table.clear(savedInst)
            if savedLighting then
                pcall(function()
                    Lighting.FogEnd = savedLighting.FogEnd
                    Lighting.FogStart = savedLighting.FogStart
                end)
                savedLighting = nil
            end
        end
        if toggleUiUpdaters["Remove Fog"] then
            toggleUiUpdaters["Remove Fog"](state)
        end
    end
end

do
    local enabled = false
    local value = 70
    local defaultFov = 70

    function Ext.setFOV(v)
        value = math.clamp(v, 1, 120)

    function Ext.toggleFOV(state)

end

local function toggleCFrameWalk(state)
    CFrameWalkEnabled = state
    if cframeConnection then 
        cframeConnection:Disconnect() 
        cframeConnection = nil 
    end
    
    if CFrameWalkEnabled then
        cframeConnection = RunService.Heartbeat:Connect(function(deltaTime)
            local char = player.Character
            if not char then return end
            local root = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChildOfClass("Humanoid")
            if root and hum and hum.MoveDirection.Magnitude > 0 then
                local speed = WalkSpeedEnabled and CustomWalkSpeed or DefaultWalkSpeed
                root.CFrame = root.CFrame + (hum.MoveDirection * speed * deltaTime)
            end
        end)
    end

local function startFlight()
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    hum.PlatformStand = true
    flyBodyVel = Instance.new("BodyVelocity")
    flyBodyVel.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    flyBodyVel.Velocity = Vector3.zero
    flyBodyVel.Parent = root

    flyBodyGyro = Instance.new("BodyGyro")
    flyBodyGyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    flyBodyGyro.CFrame = root.CFrame
    flyBodyGyro.Parent = root

    flyRenderConn = RunService.RenderStepped:Connect(function()
        if not FlightEnabled or not root or not hum then return end
        local camCFrame = camera.CFrame
        local moveDir = Vector3.zero

        if keysPressed.W then moveDir = moveDir + camCFrame.LookVector end
        if keysPressed.S then moveDir = moveDir - camCFrame.LookVector end
        if keysPressed.A then moveDir = moveDir - camCFrame.RightVector end
        if keysPressed.D then moveDir = moveDir + camCFrame.RightVector end
        if keysPressed.Up then moveDir = moveDir + Vector3.new(0, 1, 0) end
        if keysPressed.Down then moveDir = moveDir - Vector3.new(0, 1, 0) end

        if moveDir.Magnitude > 0 then moveDir = moveDir.Unit end
        flyBodyVel.Velocity = moveDir * FlySpeed
        flyBodyGyro.CFrame = camCFrame
    end)
end

local function stopFlight()
    if flyRenderConn then flyRenderConn:Disconnect() flyRenderConn = nil end
    if flyBodyVel then flyBodyVel:Destroy() flyBodyVel = nil end
    if flyBodyGyro then flyBodyGyro:Destroy() flyBodyGyro = nil end
    if player.Character and player.Character:FindFirstChildOfClass("Humanoid") then
        player.Character:FindFirstChildOfClass("Humanoid").PlatformStand = false
    end
end

local function toggleFlight(state)
    FlightEnabled = state
    FlightState = state
    if FlightEnabled then startFlight() else stopFlight() end
    if toggleUiUpdaters["Enable Flight"] then
        toggleUiUpdaters["Enable Flight"](state)
    end
end

local SpectateEnabled = false

local SpectateSaved = nil
local spectateConns = {}

local SpectateUiRefresh = nil

local function spectateDisconnectAll()
    for _, c in ipairs(spectateConns) do c:Disconnect() end
    table.clear(spectateConns)
end

local function stopSpectate(restoreCamera)
    spectateDisconnectAll()
    SpectateTarget = nil
    if restoreCamera and SpectateSaved then
        local cam = Workspace.CurrentCamera
        if cam then
            pcall(function()
                cam.CameraSubject = SpectateSaved.CameraSubject
                cam.CameraType = SpectateSaved.CameraType
            end)
        end
        SpectateSaved = nil
    end
end

local function spectateApply(target)
    local cam = Workspace.CurrentCamera
    local char = target.Character
    if not cam or not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 3)
    if not hum then return end

    spectateDisconnectAll()

    if not SpectateSaved then
        SpectateSaved = { CameraType = cam.CameraType, CameraSubject = cam.CameraSubject }
    end

    cam.CameraType = Enum.CameraType.Custom
    cam.CameraSubject = hum
    SpectateTarget = target

    table.insert(spectateConns, target.CharacterAdded:Connect(function(newChar)
        if SpectateTarget ~= target then return end
        task.spawn(function()
            local h = newChar:FindFirstChildOfClass("Humanoid") or newChar:WaitForChild("Humanoid", 5)
            if h and SpectateTarget == target then
                local c = Workspace.CurrentCamera
                if c then c.CameraSubject = h end
            end
        end)
    end))

    table.insert(spectateConns, Players.PlayerRemoving:Connect(function(p)
        if p == target then setSpectateTarget(nil) end
    end))
end

setSpectateTarget = function(target)
    if target == player then target = nil end
    if not target then
        stopSpectate(true)
        return
    end
    spectateApply(target)
end

local function toggleSpectate(state)
    SpectateEnabled = state
    if not state then
        stopSpectate(true)
    end
    if toggleUiUpdaters["Spectate Mode"] then
        toggleUiUpdaters["Spectate Mode"](state)
    end
end

local function spectateCycle(step)
    if not SpectateEnabled then return end
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then list[#list + 1] = p end
    end
    if #list == 0 then return end

    local idx = 1
    if SpectateTarget then
        for i, p in ipairs(list) do
            if p == SpectateTarget then idx = i break end
        end
    end
    idx = ((idx - 1 + step) % #list) + 1
    setSpectateTarget(list[idx])
    if SpectateUiRefresh then SpectateUiRefresh() end
end

player.CharacterAdded:Connect(function()
    if SpectateEnabled and SpectateTarget then
        task.wait(0.5)
        if SpectateEnabled and SpectateTarget then
            spectateApply(SpectateTarget)
        end
    end
end)

local function findPart(target)
    if target:IsA("BasePart") then return target end
    if target:IsA("Model") then
        return target:FindFirstChild("Head") or target.PrimaryPart or target:FindFirstChildWhichIsA("BasePart", true)
    end
    return target:FindFirstChild("Handle") or target:FindFirstChildWhichIsA("BasePart", true)
end

local function isInsideCharacter(inst)
    local m = inst:FindFirstAncestorOfClass("Model")
    return m ~= nil and m:FindFirstChildOfClass("Humanoid") ~= nil
end

-- ฟังก์ชันตรวจจับ Ghost (จาก Ghost ESP)
local GHOST_KEYWORDS = { "ghost", "monster", "entity", "demon", "killer", "nextbot" }
local GHOST_ATTRS    = { "monster", "isghost", "entity" }

local function containsAny(str, list)
    for _, k in ipairs(list) do
        if string.find(str, k, 1, true) then return true end
    end
    return false
end

local function isInsidePlayerChar(inst)
    local p = inst
    while p and p ~= Workspace do
        if Players:GetPlayerFromCharacter(p) then return true end
        p = p.Parent
    end
    return false
end

local function ghostMatch(obj)
    if not (obj:IsA("Model") or obj:IsA("BasePart")) then return false end
    if isInsidePlayerChar(obj) then return false end

    if obj:IsA("Model") and obj:FindFirstChildOfClass("AnimationController") then
        return true
    end
    if containsAny(string.lower(obj.Name), GHOST_KEYWORDS) then
        return true
    end
    for attr in pairs(obj:GetAttributes()) do
        if containsAny(string.lower(attr), GHOST_ATTRS) then return true end
    end
    return false
end

local function hasTrackedAncestor(inst)
    local p = inst.Parent
    while p and p ~= Workspace do
        if espTracked[p] then return true end
        p = p.Parent
    end
    return false
end

local function removeEntry(target)
    local e = espTracked[target]
    if not e then return end
    espTracked[target] = nil
    if e.hl then e.hl:Destroy() end
    if e.bb then e.bb:Destroy() end
end

local function createEntry(target, humanoid, isGhost)
    if espTracked[target] or not espFolder then return end

    local hl = Instance.new("Highlight")
    hl.Adornee = target
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.FillTransparency = 0.5
    hl.OutlineTransparency = 0
    hl.Enabled = false
    hl.Parent = espFolder

    local bb = Instance.new("BillboardGui")
    bb.Size = UDim2.new(0, 200, 0, 20)
    bb.AlwaysOnTop = true
    bb.StudsOffset = Vector3.new(0, humanoid and 2.5 or 1.5, 0)
    bb.Enabled = false
    bb.Parent = espFolder

    local label = Instance.new("TextLabel")
    label.Name = "Tag"
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Font = Enum.Font.GothamBold
    label.TextSize = 13
    label.TextStrokeTransparency = 0.3
    label.Parent = bb

    espTracked[target] = {
        target = target, hum = humanoid, ghost = isGhost, hl = hl, bb = bb, label = label,
        part = nil, lastText = "",
    }
end

local function classifyInstance(inst)
    if inst:IsA("Humanoid") then
        local model = inst.Parent
        if model and model:IsA("Model") and model ~= player.Character and not Players:GetPlayerFromCharacter(model) then
            return model, inst
        end
    elseif inst:IsA("ProximityPrompt") or inst:IsA("ClickDetector") then
        local target = inst.Parent
        if target and target:IsA("Attachment") then target = target.Parent end
        if target and target ~= Workspace and not isInsideCharacter(target) then
            return target, nil
        end
    elseif inst:IsA("Tool") then
        if not isInsideCharacter(inst) then return inst, nil end
    end
    return nil, nil
end

local function considerInstance(inst)
    local target, hum = classifyInstance(inst)
    if target then
        createEntry(target, hum)
        if hum and espTracked[target] and ghostMatch(target) then
            espTracked[target].ghost = true      -- ผีที่มี Humanoid
        end
    end

    if (inst:IsA("Model") or inst:IsA("BasePart")) and ghostMatch(inst) then
        if espTracked[inst] then
            espTracked[inst].ghost = true
        elseif not hasTrackedAncestor(inst) then
            createEntry(inst, nil, true)
        end
    end
end

local function refreshESP()
    if not espRunning then return end
    local camPos = camera.CFrame.Position
    local maxDist = _G.ESPMaxDistance
    local visible = {}

    for target, e in pairs(espTracked) do
        if not target:IsDescendantOf(Workspace) then
            removeEntry(target)
        else
            e.part = (e.part and e.part.Parent and e.part) or findPart(target)

            local kind, plr = "Item", nil
            if e.ghost then
                kind = "Ghost"
            elseif e.hum then
                plr = Players:GetPlayerFromCharacter(target)
                kind = plr and "Player" or "NPC"
            end

            local ignored =
                (kind == "Player" and _G.ESPIgnorePlayers) or
                (kind == "NPC"    and _G.ESPIgnoreNPCs) or
                (kind == "Ghost"  and _G.ESPIgnoreGhosts) or
                (kind == "Item"   and _G.ESPIgnoreItems) or
                plr == player

            local dist = e.part and (e.part.Position - camPos).Magnitude or math.huge

            if ignored or not e.part or dist > maxDist then
                e.hl.Enabled = false
                e.bb.Enabled = false
            else
                e.kind, e.plr, e.dist = kind, plr, dist
                visible[#visible + 1] = e
            end
        end
    end

    table.sort(visible, function(a, b) return a.dist < b.dist end)

    for i, e in ipairs(visible) do
        local color, showName, text
        if e.kind == "Player" then
            color = _G.PlayerColor
                or (_G.UseTeamColor and e.plr.TeamColor.Color
                or ((player.TeamColor == e.plr.TeamColor) and _G.FriendColor or _G.EnemyColor))
            showName = _G.ESPShowName
            text = e.plr.Name
        elseif e.kind == "Ghost" then
            color = _G.GhostColor
            showName = _G.ESPShowName
            text = "[Entity] " .. e.target.Name
        elseif e.kind == "NPC" then
            color = _G.EntityColor
            showName = _G.ESPShowName
            text = string.format("%s | HP: %d", e.target.Name, e.hum and e.hum.MaxHealth > 0 and math.floor(e.hum.Health) or 0)
        else
            color = _G.ItemColor
            showName = _G.ESPShowItemName
            text = e.target.Name
        end

        e.hl.Enabled = i <= MAX_HIGHLIGHTS
        e.hl.FillColor = color
        e.hl.OutlineColor = (e.kind == "Item" or e.kind == "Ghost") and Color3.new(1, 1, 1) or color
        e.hl.FillTransparency = 1 - _G.ESPFillOpacity
        e.hl.OutlineTransparency = 1 - _G.ESPOutlineOpacity

        e.bb.Adornee = e.part
        e.bb.Enabled = showName
        if showName then
            e.label.TextColor3 = color
            local size = (e.kind == "Item") and _G.ESPItemTextSize or _G.ESPNameTextSize
            if e.lastSize ~= size then
                e.lastSize = size
                e.label.TextSize = size
                e.bb.Size = UDim2.new(0, 300, 0, size + 8)
            end
            text = string.format("%s [%dm]", text, math.floor(e.dist))
            if text ~= e.lastText then   
                e.label.Text = text
                e.lastText = text
            end
        end
    end
end

local function setESPEnabled(state)
    if state == espRunning then return end
    espRunning = state
    espToken += 1

    for _, c in ipairs(espConns) do c:Disconnect() end
    table.clear(espConns)

    if not state then
        for target in pairs(espTracked) do removeEntry(target) end
        if espFolder then espFolder:Destroy() espFolder = nil end
        return
    end

    espFolder = Instance.new("Folder")
    espFolder.Name = "ESPContainer"
    espFolder.Parent = camera          

    table.insert(espConns, Workspace.DescendantAdded:Connect(considerInstance))
    table.insert(espConns, Workspace.DescendantRemoving:Connect(function(inst)
        if espTracked[inst] then
            removeEntry(inst)
        elseif inst:IsA("Humanoid") and inst.Parent then
            removeEntry(inst.Parent)
        end
    end))

    local myToken = espToken
    task.spawn(function()
        local count = 0
        for _, inst in ipairs(Workspace:GetDescendants()) do
            if myToken ~= espToken then return end
            considerInstance(inst)
            count += 1
            if count % 400 == 0 then task.wait() end
        end
        while myToken == espToken do
            refreshESP()
            task.wait(ESP_REFRESH_RATE)
        end
    end)
end

local function updateESPState()
    setESPEnabled(_G.ESPEnabled)
    if espRunning then refreshESP() end
end

local bypassYaw, bypassPitch = 0, 0
local bypassZoom, bypassZoomTarget = 0, 0
local bypassSaved = setmetatable({}, { __mode = "k" })   

local BODY_PARTS = {
    Head = true, Torso = true,
    ["Left Arm"] = true, ["Right Arm"] = true, ["Left Leg"] = true, ["Right Leg"] = true,
    UpperTorso = true, LowerTorso = true,
    LeftUpperArm = true, LeftLowerArm = true, LeftHand = true,
    RightUpperArm = true, RightLowerArm = true, RightHand = true,
    LeftUpperLeg = true, LeftLowerLeg = true, LeftFoot = true,
    RightUpperLeg = true, RightLowerLeg = true, RightFoot = true,
}

local bypassRayParams = RaycastParams.new()
bypassRayParams.FilterType = Enum.RaycastFilterType.Exclude
bypassRayParams.IgnoreWater = true

local function bypassSave(inst)
    if bypassSaved[inst] == nil then
        if inst:IsA("BasePart") then
            bypassSaved[inst] = { t = inst.Transparency, l = inst.LocalTransparencyModifier }
        else
            bypassSaved[inst] = { t = inst.Transparency }
        end
    end
end

local function bypassShowPart(part, hidden)
    bypassSave(part)
    pcall(function()
        part.Transparency = 0
        part.LocalTransparencyModifier = hidden and 1 or 0
    end)
    for _, d in ipairs(part:GetChildren()) do
        if d:IsA("Decal") then
            bypassSave(d)
            pcall(function() d.Transparency = 0 end)
        end
    end
end

local function bypassUpdateVisuals(char, head, firstPerson)
    for _, child in ipairs(char:GetChildren()) do
        if child:IsA("BasePart") then
            if BODY_PARTS[child.Name] then
                bypassShowPart(child, firstPerson and child == head)
            end
        elseif child:IsA("Accessory") then
            local handle = child:FindFirstChild("Handle")
            if handle and handle:IsA("BasePart") then
                bypassShowPart(handle, firstPerson and (handle.Position - head.Position).Magnitude < 1.5)
            end
        end
    end
end

local function bypassRestore()
    for inst, saved in pairs(bypassSaved) do
        if inst.Parent then
            pcall(function()
                inst.Transparency = saved.t
                if saved.l ~= nil then inst.LocalTransparencyModifier = saved.l end
            end)
        end
    end
    table.clear(bypassSaved)
end

local function toggleBypassFirstPerson(state)
    if state and not BypassFPEnabled then
        local cam = Workspace.CurrentCamera
        local head = player.Character and player.Character:FindFirstChild("Head")
        if cam then
            local rx, ry = cam.CFrame:ToOrientation()
            bypassPitch, bypassYaw = rx, ry
            bypassZoomTarget = 0
            if head then
                local d = (cam.CFrame.Position - head.Position).Magnitude
                if d > 1.5 then bypassZoomTarget = math.clamp(d, 2, BypassMaxZoom) end
            end
            bypassZoom = bypassZoomTarget
        end
    elseif not state and BypassFPEnabled then
        bypassRestore()
        pcall(function()
            local cam = Workspace.CurrentCamera
            if cam then cam.CameraType = Enum.CameraType.Custom end
        end)
        UserInputService.MouseBehavior = Enum.MouseBehavior.Default
    end

    BypassFPEnabled = state

    if toggleUiUpdaters["Bypass First Person"] then
        toggleUiUpdaters["Bypass First Person"](state)
    end
end

UserInputService.InputChanged:Connect(function(input, gpe)
    if gpe or not BypassFPEnabled then return end
    if input.UserInputType == Enum.UserInputType.MouseWheel then
        local step = 2 + bypassZoomTarget * 0.12
        bypassZoomTarget = math.clamp(bypassZoomTarget - input.Position.Z * step, 0, BypassMaxZoom)
        if bypassZoomTarget < 1.5 then bypassZoomTarget = 0 end
    end
end)

RunService:BindToRenderStep("CheckSystemBypassFP", Enum.RenderPriority.Camera.Value + 2, function(dt)
    if not BypassFPEnabled then return end

    local cam = Workspace.CurrentCamera
    local char = player.Character
    local head = char and char:FindFirstChild("Head")
    if not cam or not head then return end

    if cam.CameraType ~= Enum.CameraType.Scriptable then
        cam.CameraType = Enum.CameraType.Scriptable
    end

    bypassZoomTarget = math.min(bypassZoomTarget, BypassMaxZoom)
    bypassZoom = bypassZoom + (bypassZoomTarget - bypassZoom) * (1 - math.exp(-dt * 14))
    if math.abs(bypassZoom - bypassZoomTarget) < 0.02 then bypassZoom = bypassZoomTarget end

    local firstPersonView = bypassZoom < 1.5        
    local firstPersonMouse = bypassZoomTarget < 1.5 

    local menuOpen = MainFrame.Visible
    local rmb = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
    local rotating = rmb or (firstPersonMouse and not menuOpen)
    if UserInputService:GetFocusedTextBox() then rotating = false end   

    if rotating then
        UserInputService.MouseBehavior = (firstPersonMouse and not menuOpen)
            and Enum.MouseBehavior.LockCenter
            or Enum.MouseBehavior.LockCurrentPosition
        local delta = UserInputService:GetMouseDelta()
        local sens = math.rad(BypassSensitivity)
        bypassYaw = bypassYaw - delta.X * sens
        bypassPitch = math.clamp(bypassPitch - delta.Y * sens, -math.rad(85), math.rad(85))
    elseif UserInputService.MouseBehavior ~= Enum.MouseBehavior.Default then
        UserInputService.MouseBehavior = Enum.MouseBehavior.Default
    end

    bypassUpdateVisuals(char, head, firstPersonView)

    local rot = CFrame.fromOrientation(bypassPitch, bypassYaw, 0)
    local focus = head.Position
    local back = -rot.LookVector
    local dist = bypassZoom - 0.6
    if dist > 0 then
        bypassRayParams.FilterDescendantsInstances = { char }
        local hit = Workspace:Raycast(focus, back * dist, bypassRayParams)
        if hit then
            dist = math.max(0.4, (hit.Position - focus).Magnitude - 0.3)
        end
    end
    cam.CFrame = CFrame.new(focus + back * dist) * rot
end)

            input.Text = tostring(v)
            setSliderVisual(v)
            callback(v)
        end

        local dragging = false
        hit.InputBegan:Connect(function(io)
            if io.UserInputType == Enum.UserInputType.MouseButton1 or io.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                updateFromX(io.Position.X)
            end
        end)
        UserInputService.InputChanged:Connect(function(io)
            if dragging and (io.UserInputType == Enum.UserInputType.MouseMovement or io.UserInputType == Enum.UserInputType.Touch) then
                updateFromX(io.Position.X)
            end
        end)
        UserInputService.InputEnded:Connect(function(io)
            if io.UserInputType == Enum.UserInputType.MouseButton1 or io.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)
    end

    input.FocusLost:Connect(function()
        local val = tonumber(input.Text)
        if val then
            setSliderVisual(val)
            callback(val)
        end
    end)

    registerConfig(parent.Name .. "/" .. title,
        function() return tonumber(input.Text) end,
        function(v)
            if type(v) == "number" then
                input.Text = tostring(v)
                setSliderVisual(v)
                callback(v)
            end
        end)
end

local function parseHex(str)
    str = (tostring(str):gsub("%s", ""))
    str = (str:gsub("^#", ""))
    if #str == 3 then str = (str:gsub(".", "%0%0")) end
    if #str ~= 6 or not str:match("^%x+$") then return nil end
    return Color3.fromRGB(
        tonumber(str:sub(1, 2), 16),
        tonumber(str:sub(3, 4), 16),
        tonumber(str:sub(5, 6), 16)
    )
end

local function toHex(c)
    return string.format("%02X%02X%02X",
        math.floor(c.R * 255 + 0.5),
        math.floor(c.G * 255 + 0.5),
        math.floor(c.B * 255 + 0.5))
end

do
    local HttpService = game:GetService("HttpService")

    local CONFIG_DIR = "CheckSystem"
    local CONFIG_FOLDER = CONFIG_DIR .. "/Configs"
    local DEFAULT_FILE = CONFIG_DIR .. "/default.txt"

    local hasFS = (writefile and readfile and isfile and isfolder and makefolder and listfiles) ~= nil
    local memConfigs, memDefault = {}, nil   

    local selected = nil

    local function ensureFolders()
        if not hasFS then return end
        pcall(function()
            if not isfolder(CONFIG_DIR) then makefolder(CONFIG_DIR) end
            if not isfolder(CONFIG_FOLDER) then makefolder(CONFIG_FOLDER) end
        end)
    end

    local function sanitizeName(n)
        n = tostring(n or "")
        n = n:gsub("[^%w%s_%-]", "")
        n = n:gsub("^%s+", "")
        n = n:gsub("%s+$", "")
        return n:sub(1, 32)
    end

    local function configPath(name)
        return CONFIG_FOLDER .. "/" .. name .. ".json"
    end

    local function configExists(name)
        if hasFS then
            local ok, res = pcall(isfile, configPath(name))
            return ok and res == true
        end
        return memConfigs[name] ~= nil
    end

    local function listConfigs()
        local names = {}
        if hasFS then
            ensureFolders()
            local ok, files = pcall(listfiles, CONFIG_FOLDER)
            if ok and type(files) == "table" then
                for _, f in ipairs(files) do
                    local n = tostring(f):match("([^/\\]+)%.json$")
                    if n then names[#names + 1] = n end
                end
            end
        else
            for n in pairs(memConfigs) do names[#names + 1] = n end
        end
        table.sort(names, function(a, b) return a:lower() < b:lower() end)
        return names
    end

    local function writeConfig(name, data)
        local okEnc, json = pcall(function() return HttpService:JSONEncode(data) end)
        if not okEnc then return false, "Could not encode settings" end
        if hasFS then
            ensureFolders()
            local ok, err = pcall(writefile, configPath(name), json)
            return ok, err
        end
        memConfigs[name] = json
        return true
    end

    local function readConfig(name)
        local raw
        if hasFS then
            local ok, res = pcall(readfile, configPath(name))
            if not ok then return nil, "Could not read the config file" end
            raw = res
        else
            raw = memConfigs[name]
        end
        if not raw then return nil, "Config not found" end
        local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
        if not ok or type(data) ~= "table" then return nil, "Config file is corrupted" end
        return data
    end

    local function getDefault()
        if hasFS then
            local ok, name = pcall(function()
                if isfile(DEFAULT_FILE) then return readfile(DEFAULT_FILE) end
                return nil
            end)
            if ok and type(name) == "string" then
                name = sanitizeName(name)
                if name ~= "" and configExists(name) then return name end
            end
            return nil
        end
        if memDefault and configExists(memDefault) then return memDefault end
        return nil
    end

    local function setDefault(name)
        if hasFS then
            ensureFolders()
            local ok, err = pcall(writefile, DEFAULT_FILE, name)
            return ok, err
        end
        memDefault = name
        return true
    end

    local function collectSettings()
        local values = {}
        for _, e in ipairs(configEntries) do
            local ok, v = pcall(e.get)
            if ok and v ~= nil then values[e.key] = v end
        end
        return { version = 1, values = values }
    end

    local function applySettings(data)
        local values = data and data.values
        if type(values) ~= "table" then return false end
        for _, e in ipairs(configEntries) do
            local v = values[e.key]
            if v ~= nil then
                local ok, err = pcall(e.set, v)
                if not ok then
                    warn("[Check System] Failed to apply '" .. e.key .. "': " .. tostring(err))
                end
            end
        end
        return true
    end

    local function makeCard(height)
        local card = Instance.new("Frame", configPage)
        card.Size = UDim2.new(1, -10, 0, height)
        card.BackgroundColor3 = PURPLE_CARD
        card.BackgroundTransparency = 0.2
        card.BorderSizePixel = 0
        Instance.new("UICorner", card).CornerRadius = UDim.new(0, 8)
        return card
    end

    local nameCard = makeCard(50)
    local nameTitle = Instance.new("TextLabel", nameCard)
    nameTitle.Text = "Config Name"
    nameTitle.Font = Enum.Font.GothamBold
    nameTitle.TextSize = 13
    nameTitle.TextColor3 = TEXT_MAIN
    nameTitle.Position = UDim2.new(0, 12, 0, 8)
    nameTitle.Size = UDim2.new(0, 200, 0, 16)
    nameTitle.TextXAlignment = Enum.TextXAlignment.Left
    nameTitle.BackgroundTransparency = 1

    local nameDesc = Instance.new("TextLabel", nameCard)
    nameDesc.Text = "Name used by Save Config (new config)"
    nameDesc.Font = Enum.Font.Gotham

        messageLine.Text = text
        pcall(function()
            StarterGui:SetCore("SendNotification", { Title = "Config", Text = text, Duration = 3 })
        end)
    end

    local function updateStatus()
        statusLine.Text = "Selected: " .. (selected or "none") .. "   |   Default: " .. (getDefault() or "none")
    end

    local function rebuildList()
        for _, c in ipairs(listScroll:GetChildren()) do
            if not c:IsA("UIListLayout") and not c:IsA("UIPadding") then c:Destroy() end
        end

        local names = listConfigs()
        local default = getDefault()
        local found = false

        for _, name in ipairs(names) do
            if name == selected then found = true end
            local btn = Instance.new("TextButton", listScroll)
            btn.Size = UDim2.new(1, 0, 0, 26)
            btn.Text = "  " .. name .. (name == default and "   [DEFAULT]" or "")
            btn.Font = Enum.Font.GothamBold
            btn.TextSize = 12
            btn.TextXAlignment = Enum.TextXAlignment.Left
            btn.TextColor3 = (name == selected) and TEXT_MAIN or TEXT_SUB
            btn.BackgroundColor3 = (name == selected) and PURPLE_PRIMARY or PURPLE_CARD
            Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 5)
            btn.MouseButton1Click:Connect(function()
                selected = name
                rebuildList()
            end)
        end

        if not found then selected = nil end

        if #names == 0 then
            local empty = Instance.new("TextLabel", listScroll)
            empty.Size = UDim2.new(1, 0, 0, 26)
            empty.BackgroundTransparency = 1
            empty.Font = Enum.Font.Gotham
            empty.TextSize = 11
            empty.TextColor3 = TEXT_SUB
            empty.Text = "No configs saved yet"
        end

        updateStatus()
    end

    refreshListBtn.MouseButton1Click:Connect(rebuildList)

    addButton(configPage, "Save Config", function()
        local name = sanitizeName(nameBox.Text)
        if name == "" then
            say("Type a config name first.")
            return
        end
        if configExists(name) then
            say('"' .. name .. '" already exists. Select it and use Overwrite.')
            return
        end
        local ok, err = writeConfig(name, collectSettings())
        if ok then
            selected = name
            rebuildList()
            say('Saved config "' .. name .. '".')
        else
            say("Save failed: " .. tostring(err))
        end
    end)

    addButton(configPage, "Overwrite Config", function()
        if not selected then
            say("Select a config in the list first.")
            return
        end
        local ok, err = writeConfig(selected, collectSettings())
        if ok then
            say('Overwrote config "' .. selected .. '".')
        else
            say("Overwrite failed: " .. tostring(err))
        end
    end)

    addButton(configPage, "Load Config", function()
        if not selected then
            say("Select a config in the list first.")
            return
        end
        local data, err = readConfig(selected)
        if not data then
            say("Load failed: " .. tostring(err))
            return
        end
        if applySettings(data) then
            say('Loaded config "' .. selected .. '".')
        else
            say("Load failed: invalid config data.")
        end
    end)

    addButton(configPage, "Set Default Config", function()
        if not selected then
            say("Select a config in the list first.")
            return
        end
        local ok, err = setDefault(selected)
        if ok then
            rebuildList()
            say('"' .. selected .. '" will load automatically.')
        else
            say("Could not set default: " .. tostring(err))
        end
    end)

    rebuildList()

    loadDefaultConfig = function()
        local default = getDefault()
        if not default then return end
        local data = readConfig(default)
        if data and applySettings(data) then
            selected = default
            rebuildList()
            say('Auto-loaded default config "' .. default .. '".')
        end
    end
end

local function setupCharacter(char)
    local hum = char:WaitForChild("Humanoid")
    DefaultWalkSpeed = hum.WalkSpeed
    DefaultJumpPower = hum.JumpPower
    
    updateMovement()
    if FlightState then startFlight() end
    if NoclipState then toggleNoclip(true) end
    if CFrameWalkEnabled then toggleCFrameWalk(true) end
end

player.CharacterAdded:Connect(setupCharacter)
if player.Character then setupCharacter(player.Character) end

pages["Camera"].Visible = true
for _, child in ipairs(Sidebar:GetChildren()) do
    if child:IsA("TextButton") and child.Text:find("Camera") then
        child.TextColor3 = TEXT_MAIN
        child.BackgroundColor3 = PURPLE_PRIMARY
    end
end

UserInputService.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    
    if KeybindListening then return end   

    if input.KeyCode == Enum.KeyCode.RightShift then
        toggleGuiVisibility()
    end
    
    if input.UserInputType == Enum.UserInputType.Keyboard then
        if NoclipKey and input.KeyCode == NoclipKey then
            toggleNoclip(not NoclipState)
        end
        if FlightKey and input.KeyCode == FlightKey then
            toggleFlight(not FlightState)
        end
        if AutoTpKey and input.KeyCode == AutoTpKey then
            setAutoTeleport(not autoTpEnabled)
        end
        if FullbrightKey and input.KeyCode == FullbrightKey then
            toggleFullbright(not FullbrightEnabled)
        end
        if MasterESPKey and input.KeyCode == MasterESPKey then
            setMasterESP(not _G.ESPEnabled)
        end
        if FirstPersonKey and input.KeyCode == FirstPersonKey then
            toggleUnlockFirstPerson(not UnlockFirstPersonEnabled)
        end
        if BypassFPKey and input.KeyCode == BypassFPKey then
            toggleBypassFirstPerson(not BypassFPEnabled)
        end
    end
end)

loadDefaultConfig()
